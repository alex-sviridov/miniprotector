# listener Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** One mTLS port per host (`listener_port`, default 9300) for control-plane gRPC: a new `listener` binary verifies the client certificate and transparently proxies each call over a Unix socket to the backend serving that gRPC service on the same host.

**Architecture:** `listener` is a method-agnostic gRPC proxy (`UnknownServiceHandler` + raw-bytes codec) in a new importable package `common/svcproxy`. Backends stop opening TCP ports and call a new `connection.StartServiceServer`, which listens on `<run_dir>/services/<fully.qualified.Service>.sock`. The listener forwards the verified client certificate chain in a binary metadata header; backends re-verify it against `ca.crt` and read identity from it, so all existing role/hostname logic (`mtls.peerCert`) is unchanged.

**Tech Stack:** Go 1.26, grpc-go v1.81.1 (`encoding.CodecV2`, `mem`), testify, Docker/compose.

**Spec:** `docs/superpowers/specs/2026-10-10-listener-design.md`

## Global Constraints

- Default `listener_port` is `9300`; new config key `run_dir` (socket tree is `<run_dir>/services`). `run_dir` defaults to `<ResolveVarDir>/run`.
- Hard cutover, no dual-listen: config keys `catalog_port`, `policy_server_port`, `clientmanager_api_port`, `clientmanager_admin_api_port` and the `APIServerJobStatusPort` config field (was key `api_server_job_status_port`) are removed; the config parser rejects unknown keys, so every `local.conf` in the repo must be updated in the same change.
- Forwarded identity header: `x-mp-peer-cert-bin` (one value per certificate, leaf first, DER). The listener always overwrites it; backends reject calls without a valid one.
- Socket directory mode `0700`, socket mode `0600`. Socket file name = fully qualified gRPC service name + `.sock`; the name must match `^[A-Za-z0-9_]+(\.[A-Za-z0-9_]+)*$`.
- Error codes: no socket → `UNIMPLEMENTED` ("service <name> is not served on this host"); socket present but refusing → `UNAVAILABLE`.
- Not touched: `bwfs`, `issuer`, `step-ca`, `log-gateway`, `api-server`'s REST port (8090), all `.proto` files.
- Docs obligations (from `.claude/CLAUDE.md`): component docs, `README.md`, `docs/ARCHITECTURE.md` (incl. mermaid), `CHANGELOG.md`. No protocol-doc change (no `.proto` change).

// Package mtlstest mints throwaway CAs and certificates for tests that need
// a real mTLS identity (forwarded identity, StartServiceServer, svcproxy).
// Test-only: import it from _test.go files.
package mtlstest

import (
	"crypto/ecdsa"
	"crypto/elliptic"
	"crypto/rand"
	"crypto/tls"
	"crypto/x509"
	"crypto/x509/pkix"
	"encoding/asn1"
	"encoding/json"
	"encoding/pem"
	"math/big"
	"os"
	"path/filepath"
	"testing"
	"time"

	"github.com/stretchr/testify/require"
)

var (
	// attributeOID is the attribute extension issuer embeds on operating
	// certificates (mtls.attributeExtensionOID).
	attributeOID = asn1.ObjectIdentifier{1, 3, 6, 1, 4, 1, 61183, 1, 1}
	// issuerCallerOID marks a bootstrap-tier certificate (mtls.oidEKUIssuerCaller).
	issuerCallerOID = asn1.ObjectIdentifier{1, 3, 6, 1, 4, 1, 61183, 1, 3}
)

// CA is a throwaway certificate authority.
type CA struct {
	Cert *x509.Certificate
	Key  *ecdsa.PrivateKey
}

// LeafOption customizes a certificate minted by CA.Issue.
type LeafOption func(*x509.Certificate)

// WithAttributes embeds attrs as the JSON attribute extension.
func WithAttributes(attrs map[string]string) LeafOption {
	return func(c *x509.Certificate) {
		raw, err := json.Marshal(attrs)
		if err != nil {
			panic(err)
		}
		c.ExtraExtensions = append(c.ExtraExtensions, pkix.Extension{Id: attributeOID, Value: raw})
	}
}

// AsIssuerCaller adds the bootstrap-tier EKU.
func AsIssuerCaller() LeafOption {
	return func(c *x509.Certificate) {
		c.UnknownExtKeyUsage = append(c.UnknownExtKeyUsage, issuerCallerOID)
	}
}

// Expired makes the certificate's validity window end an hour ago.
func Expired() LeafOption {
	return func(c *x509.Certificate) {
		c.NotBefore = time.Now().Add(-2 * time.Hour)
		c.NotAfter = time.Now().Add(-time.Hour)
	}
}

func serial(t testing.TB) *big.Int {
	t.Helper()
	n, err := rand.Int(rand.Reader, big.NewInt(1<<62))
	require.NoError(t, err)
	return n
}

// NewCA creates a self-signed CA valid for an hour.
func NewCA(t testing.TB) *CA {
	t.Helper()
	key, err := ecdsa.GenerateKey(elliptic.P256(), rand.Reader)
	require.NoError(t, err)
	tmpl := &x509.Certificate{
		SerialNumber:          serial(t),
		Subject:               pkix.Name{CommonName: "test-ca"},
		NotBefore:             time.Now().Add(-time.Hour),
		NotAfter:              time.Now().Add(time.Hour),
		KeyUsage:              x509.KeyUsageCertSign | x509.KeyUsageDigitalSignature,
		BasicConstraintsValid: true,
		IsCA:                  true,
	}
	der, err := x509.CreateCertificate(rand.Reader, tmpl, tmpl, &key.PublicKey, key)
	require.NoError(t, err)
	cert, err := x509.ParseCertificate(der)
	require.NoError(t, err)
	return &CA{Cert: cert, Key: key}
}

// Issue mints a leaf for hostname (SAN + CN) usable as both client and
// server. The returned chain is [leaf, CA].
func (c *CA) Issue(t testing.TB, hostname string, opts ...LeafOption) tls.Certificate {
	t.Helper()
	key, err := ecdsa.GenerateKey(elliptic.P256(), rand.Reader)
	require.NoError(t, err)
	tmpl := &x509.Certificate{
		SerialNumber: serial(t),
		Subject:      pkix.Name{CommonName: hostname},
		DNSNames:     []string{hostname},
		NotBefore:    time.Now().Add(-time.Hour),
		NotAfter:     time.Now().Add(time.Hour),
		KeyUsage:     x509.KeyUsageDigitalSignature,
		ExtKeyUsage:  []x509.ExtKeyUsage{x509.ExtKeyUsageClientAuth, x509.ExtKeyUsageServerAuth},
	}
	for _, o := range opts {
		o(tmpl)
	}
	der, err := x509.CreateCertificate(rand.Reader, tmpl, c.Cert, &key.PublicKey, c.Key)
	require.NoError(t, err)
	return tls.Certificate{Certificate: [][]byte{der, c.Cert.Raw}, PrivateKey: key}
}

// CertsDir writes ca.crt plus identity as client.crt/client.key into a fresh
// temp directory -- the layout every miniprotector binary reads.
func (c *CA) CertsDir(t testing.TB, identity tls.Certificate) string {
	t.Helper()
	dir := t.TempDir()
	require.NoError(t, os.WriteFile(filepath.Join(dir, "ca.crt"),
		pem.EncodeToMemory(&pem.Block{Type: "CERTIFICATE", Bytes: c.Cert.Raw}), 0o600))
	var chain []byte
	for _, der := range identity.Certificate {
		chain = append(chain, pem.EncodeToMemory(&pem.Block{Type: "CERTIFICATE", Bytes: der})...)
	}
	require.NoError(t, os.WriteFile(filepath.Join(dir, "client.crt"), chain, 0o600))
	keyDER, err := x509.MarshalECPrivateKey(identity.PrivateKey.(*ecdsa.PrivateKey))
	require.NoError(t, err)
	require.NoError(t, os.WriteFile(filepath.Join(dir, "client.key"),
		pem.EncodeToMemory(&pem.Block{Type: "EC PRIVATE KEY", Bytes: keyDER}), 0o600))
	return dir
}

package mtls

import (
	"context"
	"crypto/x509"
	"fmt"

	"google.golang.org/grpc"
	"google.golang.org/grpc/codes"
	"google.golang.org/grpc/credentials"
	"google.golang.org/grpc/metadata"
	"google.golang.org/grpc/peer"
	"google.golang.org/grpc/status"
)

// ForwardedCertHeader carries the verified client certificate chain (one
// value per certificate, leaf first, DER) from the listener to a backend
// that has no TLS peer of its own. See
// docs/superpowers/specs/2026-10-10-listener-design.md.
const ForwardedCertHeader = "x-mp-peer-cert-bin"

type forwardedPeerKey struct{}

// PeerChainDER returns the DER chain the TLS peer on ctx presented (and the
// server verified), leaf first. The listener forwards it to backends.
func PeerChainDER(ctx context.Context) ([][]byte, error) {
	p, ok := peer.FromContext(ctx)
	if !ok {
		return nil, fmt.Errorf("no peer information in context")
	}
	tlsInfo, ok := p.AuthInfo.(credentials.TLSInfo)
	if !ok {
		return nil, fmt.Errorf("peer connection is not authenticated via TLS")
	}
	certs := tlsInfo.State.PeerCertificates
	if len(certs) == 0 {
		return nil, fmt.Errorf("no peer certificate presented")
	}
	chain := make([][]byte, len(certs))
	for i, c := range certs {
		chain[i] = c.Raw
	}
	return chain, nil
}

// WithForwardedChain returns a copy of md in which ForwardedCertHeader is
// replaced (never appended to) by chain, so a value the original caller
// supplied cannot survive.
func WithForwardedChain(md metadata.MD, chain [][]byte) metadata.MD {
	out := md.Copy()
	out.Delete(ForwardedCertHeader)
	for _, der := range chain {
		out.Append(ForwardedCertHeader, string(der))
	}
	return out
}

// ForwardedIdentity builds the interceptors a Unix-socket backend installs
// ahead of RequireRoles. Each call must carry ForwardedCertHeader; the chain
// is verified against certsDir/ca.crt for client auth (so a forged header from
// a local process still needs a valid certificate), a bootstrap-tier
// certificate is refused, and the leaf is stored in the context where
// peerCert finds it. Anything else is Unauthenticated.
func ForwardedIdentity(certsDir string) (grpc.UnaryServerInterceptor, grpc.StreamServerInterceptor, error) {
	roots, err := loadCAPool(certsDir)
	if err != nil {
		return nil, nil, err
	}
	resolve := func(ctx context.Context) (context.Context, error) {
		md, _ := metadata.FromIncomingContext(ctx)
		values := md.Get(ForwardedCertHeader)
		if len(values) == 0 {
			return nil, status.Error(codes.Unauthenticated, "missing forwarded peer certificate")
		}
		leaf, err := verifyForwardedChain(values, roots)
		if err != nil {
			return nil, status.Errorf(codes.Unauthenticated, "forwarded peer certificate rejected: %v", err)
		}
		return context.WithValue(ctx, forwardedPeerKey{}, leaf), nil
	}
	unary := func(ctx context.Context, req any, _ *grpc.UnaryServerInfo, handler grpc.UnaryHandler) (any, error) {
		ctx, err := resolve(ctx)
		if err != nil {
			return nil, err
		}
		return handler(ctx, req)
	}
	stream := func(srv any, ss grpc.ServerStream, _ *grpc.StreamServerInfo, handler grpc.StreamHandler) error {
		ctx, err := resolve(ss.Context())
		if err != nil {
			return err
		}
		return handler(srv, &identityStream{ServerStream: ss, ctx: ctx})
	}
	return unary, stream, nil
}

func verifyForwardedChain(values []string, roots *x509.CertPool) (*x509.Certificate, error) {
	certs := make([]*x509.Certificate, len(values))
	for i, v := range values {
		c, err := x509.ParseCertificate([]byte(v))
		if err != nil {
			return nil, fmt.Errorf("parse certificate %d: %w", i, err)
		}
		certs[i] = c
	}
	intermediates := x509.NewCertPool()
	for _, c := range certs[1:] {
		intermediates.AddCert(c)
	}
	leaf := certs[0]
	if _, err := leaf.Verify(x509.VerifyOptions{
		Roots:         roots,
		Intermediates: intermediates,
		KeyUsages:     []x509.ExtKeyUsage{x509.ExtKeyUsageClientAuth},
	}); err != nil {
		return nil, err
	}
	if hasIssuerCallerEKU(leaf) {
		return nil, fmt.Errorf("bootstrap/issuer-caller credential not accepted")
	}
	return leaf, nil
}

// identityStream overrides a ServerStream's context so handlers and the
// stream interceptors after ForwardedIdentity see the forwarded identity.
type identityStream struct {
	grpc.ServerStream
	ctx context.Context
}

func (s *identityStream) Context() context.Context { return s.ctx }

// serviceNameRe matches a fully qualified gRPC service name made of
// non-empty dot-separated identifiers -- deliberately too narrow to contain a
// path separator or "..", since the name becomes a file name.
var serviceNameRe = regexp.MustCompile(`^[A-Za-z0-9_]+(\.[A-Za-z0-9_]+)*$`)

// ValidServiceName reports whether name is safe to use as a socket file name.
func ValidServiceName(name string) bool { return serviceNameRe.MatchString(name) }

// ServicesDir is the directory holding one Unix socket per fronted service.
func ServicesDir(runDir string) string { return filepath.Join(runDir, "services") }

// ServiceSocketPath is where the backend serving gRPC service `service` listens
// and where the listener looks for it.
func ServiceSocketPath(runDir, service string) (string, error) {
	if !ValidServiceName(service) {
		return "", fmt.Errorf("invalid service name %q", service)
	}
	return filepath.Join(ServicesDir(runDir), service+".sock"), nil
}

```go
// StartServiceServer serves a backend that sits behind the listener: instead of
// a TCP port it listens on one Unix socket per gRPC service register adds
// (<runDir>/services/<ServiceName>.sock, mode 0600). There is no TLS here --
// the listener has already verified the client -- so every call must carry the
// certificate chain it forwards (mtls.ForwardedCertHeader); mtls.ForwardedIdentity
// re-verifies that chain against certsDir/ca.crt before roleRequirements are
// checked. It blocks until ctx is cancelled (returning nil) or serving fails.
func StartServiceServer(ctx context.Context, logger *slog.Logger, certsDir, runDir string, roleRequirements map[string][]string, register func(*grpc.Server), opts ...grpc.ServerOption) error {
	idUnary, idStream, err := mtls.ForwardedIdentity(certsDir)
	if err != nil {
		return fmt.Errorf("failed to build forwarded-identity check: %w", err)
	}
	roleUnary, roleStream := mtls.RequireRoles(roleRequirements)
	grpcServer := grpc.NewServer(append([]grpc.ServerOption{
		grpc.ChainUnaryInterceptor(idUnary, roleUnary),
		grpc.ChainStreamInterceptor(idStream, roleStream),
	}, append(windowServerOptions(), opts...)...)...)
	register(grpcServer)

	names := make([]string, 0)
	for name := range grpcServer.GetServiceInfo() {
		names = append(names, name)
	}
	sort.Strings(names)
	if len(names) == 0 {
		return fmt.Errorf("no gRPC services registered")
	}

	if err := os.MkdirAll(ServicesDir(runDir), 0o700); err != nil {
		return fmt.Errorf("create services dir: %w", err)
	}
	var listeners []net.Listener
	closeAll := func() {
		for _, l := range listeners {
			l.Close() // UnixListener unlinks its socket on Close
		}
	}
	for _, name := range names {
		path, err := ServiceSocketPath(runDir, name)
		if err != nil {
			closeAll()
			return err
		}
		if err := removeStaleSocket(path); err != nil {
			closeAll()
			return fmt.Errorf("service %s: %w", name, err)
		}
		l, err := net.Listen("unix", path)
		if err != nil {
			closeAll()
			return fmt.Errorf("listen on %s: %w", path, err)
		}
		listeners = append(listeners, l)
		if err := os.Chmod(path, 0o600); err != nil {
			closeAll()
			return fmt.Errorf("chmod %s: %w", path, err)
		}
		logger.Info("Service socket ready", "service", name, "socket", path)
	}

	go func() {
		<-ctx.Done()
		logger.Info("Shutting down server...")
		grpcServer.GracefulStop()
	}()

	errCh := make(chan error, len(listeners))
	for _, l := range listeners {
		go func() { errCh <- grpcServer.Serve(l) }()
	}
	var first error
	for range listeners {
		if err := <-errCh; err != nil && first == nil {
			first = err
		}
	}
	closeAll()
	return first
}

// removeStaleSocket deletes a socket file left behind by a crashed process,
// but refuses if something is still accepting on it -- a second instance of
// the same service must not steal the first one's socket.
func removeStaleSocket(path string) error {
	if _, err := os.Stat(path); errors.Is(err, os.ErrNotExist) {
		return nil
	}
	if c, err := net.DialTimeout("unix", path, 200*time.Millisecond); err == nil {
		c.Close()
		return fmt.Errorf("already running (socket %s is live)", path)
	}
	return os.Remove(path)
}
```

Note: `grpcServer.Serve` returns `nil` after `GracefulStop`, so a clean shutdown returns nil.

func unaryHandler(fullMethod string, fn func(context.Context, *wrapperspb.StringValue) (*wrapperspb.StringValue, error)) func(any, context.Context, func(any) error, grpc.UnaryServerInterceptor) (any, error) {
	return func(srv any, ctx context.Context, dec func(any) error, interceptor grpc.UnaryServerInterceptor) (any, error) {
		in := &wrapperspb.StringValue{}
		if err := dec(in); err != nil {
			return nil, err
		}
		h := func(ctx context.Context, req any) (any, error) { return fn(ctx, req.(*wrapperspb.StringValue)) }
		if interceptor == nil {
			return h(ctx, in)
		}
		return interceptor(ctx, in, &grpc.UnaryServerInfo{Server: srv, FullMethod: fullMethod}, h)
	}
}

package svcproxy

import (
	"fmt"

	"google.golang.org/grpc/mem"
)

// frame is one gRPC message carried as opaque bytes.
type frame struct{ payload []byte }

// rawCodec is an encoding.CodecV2 that moves message bytes through the proxy
// undecoded, so the proxy needs no knowledge of any service's messages. Its
// name is "proto" so the content-subtype on both hops stays
// application/grpc+proto, which every backend expects.
type rawCodec struct{}

func (rawCodec) Name() string { return "proto" }

func (rawCodec) Marshal(v any) (mem.BufferSlice, error) {
	f, ok := v.(*frame)
	if !ok {
		return nil, fmt.Errorf("svcproxy: rawCodec cannot marshal %T", v)
	}
	return mem.BufferSlice{mem.SliceBuffer(f.payload)}, nil
}

func (rawCodec) Unmarshal(data mem.BufferSlice, v any) error {
	f, ok := v.(*frame)
	if !ok {
		return fmt.Errorf("svcproxy: rawCodec cannot unmarshal into %T", v)
	}
	// data is freed once Unmarshal returns, so the payload must be copied.
	f.payload = data.Materialize()
	return nil
}

// Package svcproxy is the listener's transparent gRPC proxy: it routes each
// call, by the service named in its method path, to the backend listening on
// <runDir>/services/<Service>.sock, forwarding the caller's verified
// certificate chain. It never decodes a message, so unary and streaming RPCs
// of any service pass through and new services need no change here.
package svcproxy

import (
	"context"
	"fmt"
	"io"
	"log/slog"
	"os"
	"strings"
	"sync"
	"time"

	"github.com/alex-sviridov/miniprotector/common/connection"

	"google.golang.org/grpc"
	"google.golang.org/grpc/backoff"
	"google.golang.org/grpc/codes"
	"google.golang.org/grpc/credentials/insecure"
	"google.golang.org/grpc/metadata"
	"google.golang.org/grpc/status"
)

// Proxy forwards calls to local Unix-socket backends.
type Proxy struct {
	logger *slog.Logger
	runDir string

	mu    sync.Mutex
	conns map[string]*grpc.ClientConn // by socket path
}

// New returns a Proxy resolving backends under runDir (must be absolute).
func New(logger *slog.Logger, runDir string) *Proxy {
	return &Proxy{logger: logger, runDir: runDir, conns: map[string]*grpc.ClientConn{}}
}

// ServerOptions are the grpc.ServerOptions that make a server hand every call
// -- whatever service it names -- to this proxy, undecoded.
func (p *Proxy) ServerOptions() []grpc.ServerOption {
	return []grpc.ServerOption{
		grpc.ForceServerCodecV2(rawCodec{}),
		grpc.UnknownServiceHandler(p.handle),
	}
}

// Close releases the cached backend connections.
func (p *Proxy) Close() {
	p.mu.Lock()
	defer p.mu.Unlock()
	for path, c := range p.conns {
		c.Close()
		delete(p.conns, path)
	}
}

// ServiceFromMethod extracts the fully qualified service name from a gRPC
// full method ("/pkg.Service/Method"), rejecting anything that could not be a
// socket file name.
func ServiceFromMethod(fullMethod string) (string, error) {
	rest, ok := strings.CutPrefix(fullMethod, "/")
	if ok {
		service, method, found := strings.Cut(rest, "/")
		if found && method != "" && connection.ValidServiceName(service) {
			return service, nil
		}
	}
	return "", fmt.Errorf("malformed method %q", fullMethod)
}

func (p *Proxy) conn(sock string) (*grpc.ClientConn, error) {
	p.mu.Lock()
	defer p.mu.Unlock()
	if c, ok := p.conns[sock]; ok {
		return c, nil
	}
	c, err := grpc.NewClient("unix://"+sock,
		grpc.WithTransportCredentials(insecure.NewCredentials()),
		// A local socket: reconnect fast after a backend restart instead of
		// grpc's default backoff, which grows to two minutes.
		grpc.WithConnectParams(grpc.ConnectParams{
			Backoff:           backoff.Config{BaseDelay: 100 * time.Millisecond, Multiplier: 1.6, MaxDelay: 2 * time.Second},
			MinConnectTimeout: time.Second,
		}),
	)
	if err != nil {
		return nil, err
	}
	p.conns[sock] = c
	return c, nil
}

func (p *Proxy) handle(_ any, serverStream grpc.ServerStream) (err error) {
	start := time.Now()
	ctx := serverStream.Context()
	fullMethod, _ := grpc.MethodFromServerStream(serverStream)
	caller, _ := mtls.PeerHostname(ctx)
	defer func() {
		code := status.Code(err)
		attrs := []any{"caller", caller, "method", fullMethod, "code", code.String(), "duration", time.Since(start)}
		if code == codes.OK {
			p.logger.Debug("proxied call", attrs...)
		} else {
			p.logger.Warn("proxied call failed", attrs...)
		}
	}()

	service, err := ServiceFromMethod(fullMethod)
	if err != nil {
		return status.Error(codes.Unimplemented, err.Error())
	}
	sock, err := connection.ServiceSocketPath(p.runDir, service)
	if err != nil {
		return status.Error(codes.Unimplemented, err.Error())
	}
	if _, statErr := os.Stat(sock); statErr != nil {
		return status.Errorf(codes.Unimplemented, "service %s is not served on this host", service)
	}
	chain, err := mtls.PeerChainDER(ctx)
	if err != nil {
		return status.Errorf(codes.Unauthenticated, "no verified client certificate: %v", err)
	}
	backend, err := p.conn(sock)
	if err != nil {
		return status.Errorf(codes.Unavailable, "service %s: %v", service, err)
	}

	md, _ := metadata.FromIncomingContext(ctx)
	outCtx, cancel := context.WithCancel(metadata.NewOutgoingContext(ctx, mtls.WithForwardedChain(md, chain)))
	defer cancel()

	clientStream, err := grpc.NewClientStream(outCtx,
		&grpc.StreamDesc{ServerStreams: true, ClientStreams: true}, backend, fullMethod,
		grpc.ForceCodecV2(rawCodec{}))
	if err != nil {
		if status.Code(err) == codes.Unavailable {
			return status.Errorf(codes.Unavailable, "service %s is not accepting connections: %v", service, err)
		}
		return err
	}

	s2c := forwardServerToClient(serverStream, clientStream)
	c2s := forwardClientToServer(clientStream, serverStream)
	for i := 0; i < 2; i++ {
		select {
		case e := <-s2c:
			if e == io.EOF {
				clientStream.CloseSend()
			} else {
				cancel()
				return status.Errorf(codes.Internal, "proxying client to backend: %v", e)
			}
		case e := <-c2s:
			serverStream.SetTrailer(clientStream.Trailer())
			if e != io.EOF {
				return e // the backend's own status, passed through
			}
			return nil
		}
	}
	return status.Error(codes.Internal, "proxy loop ended unexpectedly")
}

func forwardServerToClient(src grpc.ServerStream, dst grpc.ClientStream) chan error {
	done := make(chan error, 1)
	go func() {
		f := &frame{}
		for {
			if err := src.RecvMsg(f); err != nil {
				done <- err
				return
			}
			if err := dst.SendMsg(f); err != nil {
				done <- err
				return
			}
		}
	}()
	return done
}

func forwardClientToServer(src grpc.ClientStream, dst grpc.ServerStream) chan error {
	done := make(chan error, 1)
	go func() {
		f := &frame{}
		for i := 0; ; i++ {
			if err := src.RecvMsg(f); err != nil {
				done <- err
				return
			}
			if i == 0 {
				md, err := src.Header()
				if err != nil {
					done <- err
					return
				}
				if err := dst.SendHeader(md); err != nil {
					done <- err
					return
				}
			}
			if err := dst.SendMsg(f); err != nil {
				done <- err
				return
			}
		}
	}()
	return done
}

// services: it terminates mTLS on listener_port, verifies the client
// certificate, and proxies each call over a Unix socket to the local backend
// serving that gRPC service (see common/svcproxy). A service not running on

	"syscall"

	"github.com/alex-sviridov/miniprotector/common/svcproxy"
	"google.golang.org/grpc"
)

func main() {
	const appName = "listener"

	configPath, err := config.ResolveConfigPath()
	if err != nil {
		fmt.Fprintf(os.Stderr, "Configuration error: %v\n", err)
		os.Exit(1)
	}
	conf, err := config.ParseConfig(configPath)
	if err != nil {
		fmt.Fprintf(os.Stderr, "Configuration error: %v\n", err)
		os.Exit(1)
	}

	if err != nil {
		fmt.Fprintf(os.Stderr, "Arguments error: %v\n", err)
		os.Exit(1)
	}

	ctx := logging.WithAppName(context.Background(), appName)
	ctx = context.WithValue(ctx, config.ContextKey, conf)
	ctx = logging.WithDebugMode(ctx, arguments.Debug)
	ctx = logging.WithQuietMode(ctx, false)
	logger, logfile := logging.NewLogger(ctx)
	defer logfile.Close()

		os.Exit(1)

		logger.Error("run directory resolution failed", "error", err)

		logger.Error("run directory resolution failed", "error", err)
		os.Exit(1)
